\begin{frame}{1. Application Introduction \& Problem Statement}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Motivation \& Collaboration Needs}
\footnotesize
\begin{itemize}
    \setlength{\itemsep}{3pt}
    \item \textbf{Structured Team Workspaces}: Projects require dedicated, group-isolated folders to organize code, documents, and assets without cross-group leakage.
    \item \textbf{Preventing Accidental Loss}: Public/open shared drives suffer from unauthorized deletions and unintended overwrites by arbitrary members.
    \item \textbf{Network Fragility}: Incomplete network transfers leave corrupt, unusable ghost files on the server without transactional staging.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Proposed System: Group File Sharing}
\footnotesize
\begin{itemize}
    \setlength{\itemsep}{3pt}
    \item \textbf{Client-Server Architecture}: Each group has a shared folder on the server. Files remain available when uploaders disconnect, while the server is reachable.
    \item \textbf{Group Permissions}: Members can upload, download, and create subfolders; \textbf{only Group Admins} can rename or delete files and folders.
    \item \textbf{File Transfer}: Custom LTV frames over TCP, bounded file chunks, and SHA-256 verification before publishing uploads.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{4. System Architecture: Paradigm Evaluation}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Paradigm Evaluation}
\scriptsize
\begin{itemize}
    \setlength{\itemsep}{1.5pt}
    \item \textbf{Pure P2P (Discarded)}: NAT/firewall traversal barriers; offline peers cause file unavailability; complex RBAC consensus.
    \item \textbf{Hybrid (Discarded)}: High signaling overhead; asymmetric bandwidth; weak consistency during renames.
    \item \textbf{Client-Server (Selected Architecture)}:
    \begin{itemize}
        \setlength{\itemsep}{1pt}
        \item \textbf{Authoritative RBAC}: Single source of truth for access control.
        \item \textbf{Independent of Peers}: Files stay on the server when members disconnect.
        \item \textbf{Tradeoff}: The server must be reachable and has finite storage and bandwidth.
    \end{itemize}
\end{itemize}
\end{block}
\end{column}

\begin{frame}{5. Server Engine \& Multi-Tier Architecture}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Server Component Tiers}
\scriptsize
\begin{itemize}
    \setlength{\itemsep}{2.5pt}
    \item \textbf{Connection Layer}: One worker per connection processes incoming frames in order. Outgoing frames are serialized per socket.
    \item \textbf{Session \& RBAC}: Login binds a user to the connection; each group command checks membership and role.
    \item \textbf{Concurrency}: One read/write lock per group. Members can read together; changes to shared content are exclusive.
    \item \textbf{Dual Storage Layer}: Relational SQLite (accounts, RBAC, metadata) + partitioned file tree.
\end{itemize}
\end{block}
\end{column}

\begin{frame}{6. Protocol Design: Transport Choice \& TCP Framing}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Transport Layer: TCP Byte-Stream}
\scriptsize
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{Why TCP}: Reliable, ordered byte stream with flow control. The application still handles connection failures.
    \item \textbf{Boundary Challenge}: A socket read can return part of a frame or several frames.
    \item \textbf{LTV Framing}: Read the 4-byte length first. Close the connection if $N$ is outside $1$ to $1{,}048{,}576$; otherwise read exactly $N$ more bytes.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Length-Type-Value (LTV) Frame Layout}
\centering
\vspace{2pt}
\begin{tikzpicture}[
    cell/.style={rectangle, draw=hustprimary, thick, align=center, font=\scriptsize, inner sep=2.5pt},
    headercell/.style={rectangle, draw=hustprimary, fill=hustprimary!18, thick, align=center, font=\scriptsize\bfseries, inner sep=2.5pt}
]
\node[headercell, minimum width=1.8cm] (len) {Length $N$ (4B)\\{\tiny Big-Endian}};
\node[headercell, right=0cm of len, fill=hustprimary!30, minimum width=1.2cm] (type) {Type (1B)\\{\tiny 0=JSON, 1=Bin}};
\node[cell, right=0cm of type, fill=hustprimary!6, minimum width=2.3cm] (pay) {Payload ($N-1$ B)\\{\tiny JSON or Binary Chunk}};

\draw[thick, hustprimary, decorate, decoration={brace, amplitude=3pt, mirror}] 
    ($(len.south west)+(0,-0.1)$) -- ($(type.south east)+(0,-0.1)$)
    node[midway, below=3pt, font=\tiny\bfseries] {Fixed 5-byte Header};

\draw[thick, hustprimary, decorate, decoration={brace, amplitude=3pt, mirror}] 
    ($(type.south west)+(0,-0.65)$) -- ($(pay.south east)+(0,-0.65)$)
    node[midway, below=3pt, font=\tiny\bfseries] {Length $N$ covers Type + Payload};
\end{tikzpicture}
\vspace{3pt}
\scriptsize
\begin{itemize}
    \setlength{\itemsep}{1.5pt}
    \item $N = 1 + \text{Payload size in bytes}$.
    \item \textbf{Type 0x00}: UTF-8 JSON Control / Event Message.
    \item \textbf{Type 0x01}: Raw Binary File Chunk with offset header.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\small
\begin{block}{Type 0x00: JSON Control / Event Frame}
Used for authentication, directory navigation, transfer control, and push event broadcasting.
\end{block}
\vspace{-3pt}
\centering
\begin{tikzpicture}[

\node[hdr, minimum width=2.0cm] (l1) {Length $N$ (4B)};
\node[typ, right=0cm of l1, minimum width=1.6cm] (t1) {Type = 0x00 (1B)};

\end{tikzpicture}

\vspace{3pt}
\begin{block}{Type 0x01: Raw Binary File Chunk Frame}
Carries file bytes directly, avoiding Base64 encoding.
\end{block}
\vspace{-3pt}
\centering
\begin{tikzpicture}[

]
\node[hdr, minimum width=1.6cm] (l2) {Length (4B)};
\node[typ, right=0cm of l2, minimum width=1.3cm] (t2) {Type = 0x01};
\node[sub, right=0cm of t2, minimum width=2.2cm] (u2) {Transfer UUID (16B)};
\node[sub, right=0cm of u2, minimum width=1.5cm] (o2) {Offset (8B)};
\node[sub, right=0cm of o2, minimum width=1.5cm] (c2) {ChunkLen (4B)};
\node[raw, right=0cm of c2, minimum width=3.8cm] (d2) {Raw File Bytes (ChunkLen)};
\end{tikzpicture}
\vspace{2pt}
\begin{itemize}
    \scriptsize
    \item \textbf{Transfer UUID (16 Bytes)}: Identifies the active upload/download staging session.
    \item \textbf{Offset (8 Bytes)}: Must match the next expected byte; starts at zero.
    \item \textbf{ChunkLen (4 Bytes)}: 1 to 32 KiB of raw bytes; $N = 29 + \texttt{ChunkLen}$. Integers use big-endian encoding.
\end{itemize}
\end{frame}

\begin{frame}{13. State Machine: Resilient File Transfer}
\small
\begin{block}{Staged File Upload Lifecycle: Preventing Corrupted Transfers}
Directly resolves the challenge of broken TCP connections leaving partial ghost files on disk.
\end{block}
\vspace{-2pt}
\centering
\begin{tikzpicture}[
    st/.style={rectangle, rounded corners=3pt, draw=hustprimary, fill=hustprimary!10, thick, font=\scriptsize\bfseries, align=center, inner sep=3pt},
    err/.style={rectangle, rounded corners=3pt, draw=red!80!black, fill=red!10, thick, font=\scriptsize\bfseries, align=center, inner sep=3pt},
    arr/.style={-{Latex[length=1.5mm, width=1.1mm]}, thick, color=hustprimary, font=\tiny},
    errarr/.style={-{Latex[length=1.5mm, width=1.1mm]}, thick, color=red!80!black, dashed, font=\tiny}
]
% Row 1
\node[st] (idle) {IDLE};
\node[st, right=2.0cm of idle] (rcv) {RECEIVING CHUNKS\\{\tiny Write to private staging file}};
\node[st, right=2.2cm of rcv] (vfy) {VERIFY INTEGRITY\\{\tiny Check received size and SHA-256}};

% Row 2
\node[err, below=1.25cm of rcv] (abt) {ABORTED\\{\tiny Discard staging file}};
\node[st, below=1.25cm of vfy] (cmt) {COMMITTED\\{\tiny File and metadata ready}};

% Success transitions
\draw[arr] (idle) -- node[above=2pt] {\texttt{FILE\_UPLOAD\_INIT}} (rcv);
\draw[arr] (rcv) -- node[above=2pt] {\texttt{FILE\_UPLOAD\_FINISH}} (vfy);
\draw[arr] (vfy) -- node[right=3pt] {Checks and commit succeed} (cmt);

% Failure transitions
\draw[errarr] (rcv) -- node[left=2pt] {\shortstack[l]{Abort, timeout,\\disconnect, write failure}} (abt);
\draw[errarr] (vfy.south west) -- node[above=2pt, sloped] {Verification or commit fails} (abt.north east);
\end{tikzpicture}
\vspace{3pt}
\scriptsize
\begin{itemize}
    \setlength{\itemsep}{1.5pt}
    \item \textbf{Receiving}: Each accepted chunk advances the expected offset; receiving all bytes does not publish the file.
    \item \textbf{Commit}: Recheck permissions and destination under the group write lock. Move the file on the same filesystem and record its metadata before replying.
    \item \textbf{Recovery}: Failed uploads stay out of group listings. On startup, reconcile staging files and files moved before their metadata was recorded.
\end{itemize}
\end{frame}

\begin{frame}{14. Message Processing \& Concurrency Pipeline}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Server Pipeline Workflow}
\scriptsize
\begin{enumerate}
    \setlength{\itemsep}{2pt}
    \item \textbf{Decode}: Check length and type. Close on invalid framing.
    \item \textbf{Validate}: Check command fields and state. For chunks, check transfer ownership, expected offset, and remaining size.
    \item \textbf{Authorize}: Check the user's role in the target group. Reject paths outside that group's root.
    \item \textbf{Execute}: Take the required group lock and perform the operation.
    \item \textbf{Reply}: Return a result or error. Send change events after success.
\end{enumerate}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Locking Architecture: Solving Race Conditions}
\centering
\vspace{1pt}
\begin{tikzpicture}[
    box/.style={rectangle, rounded corners=2pt, draw=hustprimary, fill=hustprimary!8, thick, font=\tiny, align=center, inner sep=2.5pt},
    arr/.style={-{Latex[length=1.5mm, width=1.1mm]}, thick, color=hustprimary}
]
\node[box, text width=6.2cm] (b1) {\textbf{Member 1: Download} \texttt{/reports/A.pdf}\\Takes the group's shared read lock};
\node[box, below=0.2cm of b1, text width=6.2cm] (b2) {\textbf{Member 2: Download} \texttt{/reports/A.pdf}\\Takes the same shared read lock};
\node[box, below=0.2cm of b2, text width=6.2cm, fill=red!10, draw=red!80!black] (b3) {\textbf{Admin: Delete / Rename} \texttt{/reports/}\\Tries the group's exclusive write lock\\Receives \texttt{LOCK\_CONFLICT}; may retry later};

\draw[arr] (b1) -- (b2);
\draw[arr] (b2) -- (b3);
\end{tikzpicture}
\vspace{2pt}
\begin{itemize}
    \tiny
    \item The download worker releases its lock after the last write or on failure. A group lock can also block writes to unrelated folders in that group.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}
