% Slide 1: Actors & Problem Definition

\begin{frame}{1. Actors \& Problem Definition}
\small
\begin{columns}[T]
\begin{column}{0.43\textwidth}
\begin{block}{System Actors}
\begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \textbf{Regular Member}:
    \begin{itemize}
        \item Registers \& signs in to the system.
        \item Requests to join groups.
        \item Uploads/downloads files, creates subfolders in group space.
    \end{itemize}
    \item \textbf{Group Admin} (Group Creator):
    \begin{itemize}
        \item Creates and manages sharing groups.
        \item Exclusive rights: rename \& delete files and subfolders.
    \end{itemize}
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.53\textwidth}
\begin{block}{Problem Definition}
\begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \textbf{Group-Centric Collaboration}: Provide dedicated shared folders for teams to organize and collaborate online.
    \item \textbf{Hierarchical Content Organization}: Allow members to structure project data with nested subfolders and file uploads.
    \item \textbf{Role-Based Access Control}: Enforce administrative authority so that only group admins can rename or delete files and folders, preventing unauthorized data loss.
    \item \textbf{Network File Transfer}: Support secure client-server file uploads and downloads across the network.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 2: Core Use Cases

\begin{frame}{2. Core Use Cases}
\small
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{General Use Cases}
\begin{itemize}
    \setlength{\itemsep}{5pt}
    \item \textbf{UC1 --- Register \& Log In}\\
    Create an account and authenticate for usage.
    \item \textbf{UC2 --- Create or Join Group}\\
    Create a new group or request to join existing ones.
    \item \textbf{UC3 --- Upload \& Download Files}\\
    Upload files to group storage and download shared files.
    \item \textbf{UC4 --- Create Subfolders}\\
    Create subdirectories within the group folder.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Admin-Only Use Cases}
\begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \textbf{UC5 --- Delete Files \& Subfolders}\\
    Exclusive authority to remove files and nested folders within the group space.
    \item \textbf{UC6 --- Rename Files \& Folders}\\
    Exclusive authority to rename files and directories.
    \item \textbf{UC7 --- Manage Group Membership}\\
    Review, accept, or reject user requests to join the group.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 3: Inputs & Outputs Specification

\begin{frame}{3. Inputs \& Outputs Specification}
\footnotesize
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{General Operations}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{UC1 --- Register \& Log In}\\
    $\bullet$ \textit{In:} Username, password credentials.\\
    $\bullet$ \textit{Out:} Auth status, session token.
    \item \textbf{UC2 --- Create or Join Group}\\
    $\bullet$ \textit{In:} Group info (create) / target ID (join).\\
    $\bullet$ \textit{Out:} Group folder initialized / request status.
    \item \textbf{UC3 --- Upload \& Download Files}\\
    $\bullet$ \textit{In:} Target path, stream (up) / file path (down).\\
    $\bullet$ \textit{Out:} Stored file notice / file stream.
    \item \textbf{UC4 --- Create Subfolders}\\
    $\bullet$ \textit{In:} Parent directory, subfolder name.\\
    $\bullet$ \textit{Out:} Subfolder created, updated directory.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{Admin-Only Operations}
\begin{itemize}
    \setlength{\itemsep}{5pt}
    \item \textbf{UC5 --- Delete Files \& Subfolders}\\
    $\bullet$ \textit{In:} Admin token, target item path.\\
    $\bullet$ \textit{Out:} Permission status, item deleted, updated view.
    \item \textbf{UC6 --- Rename Files \& Folders}\\
    $\bullet$ \textit{In:} Admin token, item path, new name.\\
    $\bullet$ \textit{Out:} Validation check, rename confirmation.
    \item \textbf{UC7 --- Manage Group Membership}\\
    $\bullet$ \textit{In:} Admin token, request ID, decision.\\
    $\bullet$ \textit{Out:} Updated member list, notification.
\end{itemize}
\end{block}
\end{column}
\end{columns}
