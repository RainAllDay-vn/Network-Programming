% Slide 1: Problem Definition & Use Cases
% -----------------------------------------------------------------------------
\begin{frame}{1. Problem Definition \& Core Use Cases}
\footnotesize
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Core Use Cases}
\begin{itemize}
    \setlength{\itemsep}{1.5pt}
    \item \textbf{UC1 (Account Creation)}: Create user account and authenticate.
    \item \textbf{UC2 (Publish Files)}: Announce local files in the catalog.
    \item \textbf{UC3 (Cancel Publication)}: Withdraw files from the catalog.
    \item \textbf{UC4 (Search by Name)}: Search files and view hosting machines.
    \item \textbf{UC5 (Download File)}: Select a machine and download directly.
    \item \textbf{UC6 (Report Error)}: Report failed downloads for catalog cleanup.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{System Actors}
\textbf{File Publisher}: Makes local files available for sharing.\\
\textbf{File Seeker}: Searches and downloads files directly.
\end{block}
\vspace{-0.3em}
\begin{block}{Problem Definition}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{Resource Discovery \& Access}: Locate which active machines hold desired files, access authorized shares, and maintain an accurate catalog as machines join and leave.
    \item \textbf{Direct Peer-to-Peer Transfer}: Establish direct machine-to-machine connections to stream file data reliably and with integrity, without any intermediary storage server.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 2: Inputs & Outputs Specification
% -----------------------------------------------------------------------------
\begin{frame}{2. Inputs \& Outputs Specification}
\footnotesize
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Account \& Catalog Operations}
\begin{itemize}
    \setlength{\itemsep}{3pt}
    \item \textbf{Account Creation \& Login}:\\
    $\bullet$ \textit{In:} Username, password credentials.\\
    $\bullet$ \textit{Out:} Auth status (success/failure), session state.
    \item \textbf{File Publication}:\\
    $\bullet$ \textit{In:} File name, metadata, host machine address.\\
    $\bullet$ \textit{Out:} Publication confirmation; catalog index updated.
    \item \textbf{Cancel Publication}:\\
    $\bullet$ \textit{In:} Target file identifier to revoke.\\
    $\bullet$ \textit{Out:} Revocation status; file removed from catalog.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{Search, Download \& Maintenance}
\begin{itemize}
    \setlength{\itemsep}{3pt}
    \item \textbf{Search for Files}:\\
    $\bullet$ \textit{In:} Target file name / keyword query.\\
    $\bullet$ \textit{Out:} List of machine addresses hosting matching files.
    \item \textbf{Direct File Download}:\\
    $\bullet$ \textit{In:} Selected machine address, target file name.\\
    $\bullet$ \textit{Out:} File content streamed to local storage.
    \item \textbf{Report Indexing Error}:\\
    $\bullet$ \textit{In:} Unsuccessful download signal (failed host/file).\\
    $\bullet$ \textit{Out:} Error logged; catalog updated to purge entry.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 3: Functional & Non-Functional Requirements
% -----------------------------------------------------------------------------
\begin{frame}{3. Functional \& Non-Functional Requirements}
\footnotesize
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Functional Requirements (FR)}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{FR1 (Account Management)}: Support user account creation and login.
    \item \textbf{FR2 (File Publishing)}: Enable users to publish available files and cancel publications.
    \item \textbf{FR3 (Search \& Lookup)}: Search files by name; return addresses of machines holding those files.
    \item \textbf{FR4 (Direct Download)}: Allow users to retrieve files directly from a designated machine.
    \item \textbf{FR5 (Catalog Self-Correction)}: Receive error reports for failed downloads and update file catalog.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{Non-Functional Requirements (NFR)}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \item \textbf{Data Integrity}: Files transferred directly between machines must remain bit-for-bit identical to the source.
    \item \textbf{Catalog Accuracy}: The file index must accurately reflect available files and prune invalid entries.
    \item \textbf{Fault Resilience}: Transfer failures or offline machines must be handled gracefully with clear user feedback.
    \item \textbf{Controlled Privacy}: Only files explicitly published by the user are accessible to other network peers.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}
