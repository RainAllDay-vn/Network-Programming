\begin{frame}{1. Actors \& Problem Definition}
\small
\begin{columns}[T]
\begin{column}{0.43\textwidth}
\begin{block}{System Actors}
\begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \textbf{Regular Member}:
    \begin{itemize}
        \item Registers \& signs in to the system.
        \item Requests to join groups.
        \item Uploads/downloads files, creates subfolders in group space.
    \end{itemize}
    \item \textbf{Group Admin} (Group Creator):
    \begin{itemize}
        \item Creates and manages sharing groups.
        \item Exclusive rights: rename \& delete files and subfolders.
    \end{itemize}
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.53\textwidth}
\begin{block}{Problem Definition}
\begin{itemize}
    \setlength{\itemsep}{6pt}
    \item \textbf{Group-Centric Collaboration}: Provide dedicated shared folders for teams to organize and collaborate online.
    \item \textbf{Hierarchical Content Organization}: Allow members to structure project data with nested subfolders and file uploads.
    \item \textbf{Role-Based Access Control}: Enforce administrative authority so that only group admins can rename or delete files and folders, preventing unauthorized data loss.
    \item \textbf{Network File Transfer}: Support secure client-server file uploads and downloads across the network.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------

\begin{frame}{2. Core Use Cases}
\small
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Account \& Group Management}
\begin{itemize}
    \setlength{\itemsep}{8pt}
    \item \textbf{UC1 --- Register \& Log In}\\
    Register a new account and authenticate for usage.
    \item \textbf{UC2 --- Create Sharing Group}\\
    Create a new group with a dedicated shared folder; creator gains admin privileges.
    \item \textbf{UC3 --- Join Group}\\
    Browse existing groups and submit requests to join.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{File \& Folder Operations}
\begin{itemize}
    \setlength{\itemsep}{8pt}
    \item \textbf{UC4 --- Upload \& Download Files}\\
    Any group member can upload files to the group directory or download files to local storage.
    \item \textbf{UC5 --- Create Subfolders}\\
    Group members can create subfolders inside the group's shared folder.
    \item \textbf{UC6 --- Manage Files \& Folders (Admin)}\\
    Exclusive authority to delete files/subfolders and rename files/folders.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 3: Inputs & Outputs Specification

\begin{frame}{3. Inputs \& Outputs Specification}
\footnotesize
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Account \& Group Operations}
\begin{itemize}
    \setlength{\itemsep}{3pt}
    \item \textbf{Account Registration \& Login}:\\
    $\bullet$ \textit{In:} Username, password credentials.\\
    $\bullet$ \textit{Out:} Auth status (success/failure), session token.
    \item \textbf{Create Sharing Group}:\\
    $\bullet$ \textit{In:} Group name, creator user ID.\\
    $\bullet$ \textit{Out:} Group created, dedicated folder provisioned, admin privileges assigned.
    \item \textbf{Request to Join Group}:\\
    $\bullet$ \textit{In:} Target Group ID, requesting user ID.\\
    $\bullet$ \textit{Out:} Join request status (pending/approved), member list updated.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{File \& Directory Operations}
\begin{itemize}
    \setlength{\itemsep}{3pt}
    \item \textbf{File Upload \& Download}:\\
    $\bullet$ \textit{In:} Group/folder path, file stream/metadata (upload); file path (download).\\
    $\bullet$ \textit{Out:} File saved / streamed to client; transfer status.
    \item \textbf{Create Subfolder}:\\
    $\bullet$ \textit{In:} Target parent path, new subfolder name.\\
    $\bullet$ \textit{Out:} Subfolder created; directory view updated.
    \item \textbf{Rename \& Delete (Admin Only)}:\\
    $\bullet$ \textit{In:} Admin session, target item path, new name (rename) or deletion signal.\\
    $\bullet$ \textit{Out:} Permission validated; item renamed or deleted; updated tree returned.
\end{itemize}
\end{block}
\end{column}
\end{columns}
