% Slide 1: Actors & Problem Definition

\begin{frame}{1. Actors \& Problem Definition}
\small
\begin{columns}[T]
\begin{column}{0.40\textwidth}
\begin{block}{System Actors}
\begin{itemize}
    \setlength{\itemsep}{8pt}
    \item \textbf{File Publisher}: Makes selected local files available to others.
    \item \textbf{File Seeker}: Searches for files and downloads them from other users.
\end{itemize}
\medskip
The same user can take both roles.
\end{block}
\end{column}
\begin{column}{0.56\textwidth}
\begin{block}{Problem Definition}
\begin{itemize}
    \setlength{\itemsep}{8pt}
    \item \textbf{Find shared files}: Help users locate machines that offer the files they need.
    \item \textbf{Transfer files directly}: Download from the machine holding the file, without intermediary file storage.
    \item \textbf{Keep listings accurate}: Update the catalog when files are withdrawn or downloads fail.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 2: Core Use Cases
% -----------------------------------------------------------------------------
\begin{frame}{2. Core Use Cases}
\small
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Account \& File Publication}
\begin{itemize}
    \setlength{\itemsep}{10pt}
    \item \textbf{UC1 --- Register \& Log In}\\
    Create an account and sign in.
    \item \textbf{UC2 --- Publish Files}\\
    List local files for others to find.
    \item \textbf{UC3 --- Cancel Publication}\\
    Remove a file from the shared catalog.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{File Search \& Download}
\begin{itemize}
    \setlength{\itemsep}{10pt}
    \item \textbf{UC4 --- Search by Name}\\
    Find files and view machines offering them.
    \item \textbf{UC5 --- Download a File}\\
    Select a machine and download directly.
    \item \textbf{UC6 --- Report a Failed Download}\\
    Report the failure so the catalog can be updated.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 3: Inputs & Outputs Specification
% -----------------------------------------------------------------------------
\begin{frame}{3. Inputs \& Outputs Specification}
\footnotesize
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Account \& Catalog Operations}
\begin{itemize}
    \setlength{\itemsep}{3pt}
    \item \textbf{Account Creation \& Login}:\\
    $\bullet$ \textit{In:} Username, password credentials.\\
    $\bullet$ \textit{Out:} Auth status (success/failure), session state.
    \item \textbf{File Publication}:\\
    $\bullet$ \textit{In:} File name, metadata, host machine address.\\
    $\bullet$ \textit{Out:} Publication confirmation; catalog index updated.
    \item \textbf{Cancel Publication}:\\
    $\bullet$ \textit{In:} Target file identifier to revoke.\\
    $\bullet$ \textit{Out:} Revocation status; file removed from catalog.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{Search, Download \& Maintenance}
\begin{itemize}
    \setlength{\itemsep}{3pt}
    \item \textbf{Search for Files}:\\
    $\bullet$ \textit{In:} Target file name / keyword query.\\
    $\bullet$ \textit{Out:} List of machine addresses hosting matching files.
    \item \textbf{Direct File Download}:\\
    $\bullet$ \textit{In:} Selected machine address, target file name.\\
    $\bullet$ \textit{Out:} File content streamed to local storage.
    \item \textbf{Report Indexing Error}:\\
    $\bullet$ \textit{In:} Unsuccessful download signal (failed host/file).\\
    $\bullet$ \textit{Out:} Error logged; catalog updated to purge entry.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}
