\Large Building a File Sharing Application

\begin{block}{Project idea}
Develop a networked application where users can securely organize and share files in group workspaces.
\end{block}
\begin{block}{Goal}
Combine account and group management with reliable file transfer and controlled collaboration.

\begin{frame}{Sharing Within Groups}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Member workflow}
\begin{itemize}
\item Create an account and sign in to the application.
\item Create a group or request access to an existing group; members may also be invited.
\item Once approved, members use the group's shared folder to exchange files.
\item Upload and download files, browse contents, and organize work into subfolders.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Group roles}
\begin{itemize}
\item The group creator manages the group as its administrator.
\item Members contribute files and create subfolders.
\item The administrator manages membership and controls renaming or deleting shared files and folders.
\item Members can organize shared content by copying or moving files and folders.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Proposed Network Design}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Request handling}
\begin{itemize}
\item Users connect to a central server using socket-based communication.
\item The server authenticates sessions and checks group membership and permissions before processing requests.
\item Group actions and file operations are recorded for activity tracking.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{File service}
\begin{itemize}
\item Stream file data between client and server to support large uploads and downloads.
\item Keep each group's content in its own shared folder, with tools to browse and organize files and subfolders.
\item Apply role-based permissions to file and folder changes.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{0.5em}
\centering\textbf{Client} $\longleftrightarrow$ \textbf{Socket server} $\longleftrightarrow$ \textbf{Group storage}
\end{frame}
